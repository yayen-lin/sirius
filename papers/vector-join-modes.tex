    % row 2: native SQL
    \vjrow{Native SQL}%
    \begin{minipage}[c]{\dimexpr\textwidth-1.2em\relax}
        \begin{tcbraster}[raster columns=3, raster equal height=rows, raster before skip=1mm, raster after skip=0pt]
            \begin{tcolorbox}[vjcard, vjmode=vjperrow]
                \begin{lstlisting}[style=vj,label={lst:sql-per-row}]
SELECT R.id, nn.id, nn.dist
FROM R, @LATERAL@ (
  SELECT S.id,
    array_distance(R.emb, S.emb) AS dist
  FROM S
  @ORDER BY array_distance(R.emb, S.emb)@
  @LIMIT 5@) nn;
\end{lstlisting}
            \end{tcolorbox}
            \begin{tcolorbox}[vjcard, vjmode=vjglobal]
                \begin{lstlisting}[style=vj,label={lst:sql-global}]
SELECT R.id, S.id,
  array_distance(R.emb, S.emb) AS dist
FROM R, S
@ORDER BY dist LIMIT 5@;
\end{lstlisting}
            \end{tcolorbox}
            \begin{tcolorbox}[vjcard, vjmode=vjthresh]
                \begin{lstlisting}[style=vj,label={lst:sql-threshold}]
SELECT R.id, S.id,
  array_distance(R.emb, S.emb) AS dist
FROM R JOIN S
  @ON array_distance(R.emb, S.emb) <= 5@;
\end{lstlisting}
            \end{tcolorbox}
        \end{tcbraster}
    \end{minipage}
