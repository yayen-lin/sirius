sirius_knn_join(
  left_table,    -- probe table R
  left_column,   -- vector column of left table
  right_table,   -- corpus table S
  right_column,  -- vector column of right table
  k =>,          -- result size for top-k
  eps =>,        -- distance bound for threshold
  metric => ['l2' | 'cosine'],
  join_mode => ['global' | 'per-row' | 'threshold'],
  output_type => ['similarity' | 'distance'],
  search_mode => ['approx' | 'exact' | 'exact-gemm'],
  left_output_columns =>,  -- default to pk/id (fast path)
  right_output_columns =>  -- default to pk/id (fast path)
)
\end{lstlisting}

% ---- per-row top-k ----
    \colorlet{vjmode}{vjperrow}
    \begin{tcolorbox}
                [vjcard, title={Per-row top-$k$}]
            \begin{lstlisting}[style=vj,label={lst:tvf-per-row}]
SELECT * FROM sirius_knn_join('R', 'emb', 'S', 'emb',
  @join_mode => 'per-row', k => 5@, metric => 'l2');
\end{lstlisting}
            \tcblower
            \begin{lstlisting}[style=vj,label={lst:sql-per-row}]
SELECT R.id, nn.id, nn.dist
FROM R, @LATERAL@ (
  SELECT S.id, array_distance(R.emb, S.emb) AS dist FROM S
  @ORDER BY array_distance(R.emb, S.emb) LIMIT 5@) nn;
\end{lstlisting}
    \end{tcolorbox}
