% ---- per-row top-k ----
\colorlet{vjmode}{vjperrow}
\begin{tcolorbox}[vjcard, title={\bfseries Per-row top-$k$\hfill\itshape\mdseries for each row of $R$, its $k$ nearest rows of $S$}]
\begin{lstlisting}[style=vj]
SELECT * FROM sirius_knn_join(
  'R', 'emb', 'S', 'emb',
  @join_mode => 'per-row', k => 5@,
  metric => 'l2');
\end{lstlisting}
\tcblower
\begin{lstlisting}[style=vj]
SELECT R.id, nn.id, nn.dist
FROM R, @LATERAL@ (
  SELECT S.id, array_distance(R.emb, S.emb) AS dist
  FROM S
  @ORDER BY array_distance(R.emb, S.emb) LIMIT 5@) nn;
\end{lstlisting}
\end{tcolorbox}

% ---- global top-k ----
\colorlet{vjmode}{vjglobal}
\begin{tcolorbox}[vjcard, title={\bfseries Global top-$k$\hfill\itshape\mdseries the $k$ closest pairs in $R \times S$}]
\begin{lstlisting}[style=vj]
SELECT * FROM sirius_knn_join(
  'R', 'emb', 'S', 'emb',
  @join_mode => 'global', k => 5@,
  metric => 'l2');
\end{lstlisting}
\tcblower
\begin{lstlisting}[style=vj]
SELECT R.id, S.id,
       array_distance(R.emb, S.emb) AS dist
FROM R, S
@ORDER BY dist LIMIT 5@;
\end{lstlisting}
\end{tcolorbox}

% ---- threshold ----
\colorlet{vjmode}{vjthresh}
\begin{tcolorbox}[vjcard, title={\bfseries Threshold\hfill\itshape\mdseries every pair within distance $\varepsilon$}]
\begin{lstlisting}[style=vj]
SELECT * FROM sirius_knn_join(
  'R', 'emb', 'S', 'emb',
  @join_mode => 'threshold', eps => 5@,
  metric => 'l2');
\end{lstlisting}
\tcblower
\begin{lstlisting}[style=vj]
SELECT R.id, S.id,
       array_distance(R.emb, S.emb) AS dist
FROM R JOIN S
  @ON array_distance(R.emb, S.emb) <= 5@;
\end{lstlisting}
\end{tcolorbox}
